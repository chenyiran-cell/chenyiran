📖 核心代码（async/await + Promise.race + Promise.all）
function delay(ms, label) { return new Promise(resolve => { setTimeout(() => resolve(label), ms); }); } async function makeDumplings() { // 1️⃣ Promise.race：同时线上线下买，谁先完成用谁 const bought = await Promise.race([ delay(6000, '线上超市'), delay(4000, '菜市场') ]); // 2️⃣ 取原料 await delay(2000, '取原料'); // 3️⃣ Promise.all：和面 + 和馅 并行，两个都完成才继续 const [dough, filling] = await Promise.all([ delay(5000, '和面'), delay(3000, '和馅') ]); // 4️⃣ 包饺子（串行） await delay(8000, '包饺子'); // 5️⃣ 煮饺子 await delay(4000, '煮饺子'); // 6️⃣ 吃饺子！ return await delay(3000, '吃光光 🎉'); } makeDumplings().then(res => console.log(res));
